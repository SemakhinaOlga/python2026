def extract_numbers(text):
    # Удаляем основные знаки препинания (можно расширить)
    for punct in '.,–!?;:“”':
        text = text.replace(punct, '')
    words = text.split()
    # Используем filter для отбора чисел
    numbers = list(filter(lambda w: w.isdigit(), words))
    # Считаем частоты
    freq = {}
    for num in numbers:
        freq[num] = freq.get(num, 0) + 1
    # Формируем результат с порядковыми номерами
    result = [(i, numbers[i], freq[numbers[i]]) for i in range(len(numbers))]
    return result

# Пример использования
